% 方块里的字。按原书用语与大小写填。

\newcommand{\exbanner}[2]{%
  \par\needspace{6\baselineskip}\medskip
  \noindent
  \ifStyleBanner
    % 先画细线，再画不透明的方块把它压住
    \rlap{\raisebox{\StyleBannerRaise}{\rlap{\color{StyleBannerRule}%
      \rule{\linewidth}{0.5pt}}}}%
    \colorbox{StyleBanner}{\textcolor{StyleBannerText}{\StyleSansFont\bfseries
      \fontsize{14}{16}\selectfont\,\StyleBannerLabel\,}}%
    \hspace{0.55em}%
    {\StyleSansFont\bfseries\fontsize{17}{18}\selectfont\color{StyleNumber}#1}%
    \ifx\relax#2\relax\else\hspace{0.7em}{\StyleSansFont\bfseries\large\color{StyleAccent}#2}\fi
  \else
    {\StyleSansFont\bfseries\large\color{StyleAccent}#1}%
    \ifx\relax#2\relax\else\quad{\StyleSansFont\large\color{StyleAccent}#2}\fi
    \par\smallskip
    \noindent\textcolor{StyleBannerRule}{\rule{\linewidth}{0.5pt}}%
  \fi
  \par\medskip
}

x & f(x) & g(x) & f'(x) & g'(x)\\[2pt]
