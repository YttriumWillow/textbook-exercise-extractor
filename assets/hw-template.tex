% ---- 2. palette ---------------------------------------------------------------------
%   book: measured from the exercise pages
\definecolor{StyleInk}{HTML}{231F20}        % body text
\definecolor{StyleAccent}{HTML}{0089CF}     % blue sans group headings ("Economics")
\definecolor{StyleBanner}{HTML}{00608A}     % filled EXERCISES banner

\definecolor{StyleBannerRule}{HTML}{939598} % grey hairline the banner hangs from
\definecolor{StyleHeadText}{HTML}{231F20}   % running-head text
\definecolor{StyleHeadRule}{HTML}{ED1846}   % running-head hairline (unused when off)

% ---- 3. banner ----------------------------------------------------------------------
%   In the source this is: a full-width grey hairline, with a filled box carrying white
%   "EXERCISES" hanging from it, then the section number in dark crimson beside it.
%   Some books have no banner (this one has: label, box 79x22pt). If yours does not, the
%   probe proposes \StyleBannerfalse and the section heading degrades to a plain
%   coloured line with a hairline under it.
\newif\ifStyleBanner
\StyleBannertrue

%   Vertical offset from the banner's text baseline up to the hairline. This value is the
%   template's own box geometry (0.75 x the 14pt label below + 3pt \fboxsep), tuned so the
%   hairline sits on the box's top edge -- the probe reports the book's own number as a
%   comment (15.3pt between baseline and rule for this book). Nudge if the rule cuts the box.
\newcommand{\StyleBannerRaise}{13.5pt}

%   The word printed inside the box. Match the book's own language and case.
\newcommand{\StyleBannerLabel}{EXERCISES}

% ---- 4. instructions -----------------------------------------------------------------
%   Measured: this book sets the exercise-set instruction in upright roman, same size as
%   the body -- the blue group heading above it is what separates it. Set to \relax for
%   that; set to \itshape if your book italicises instructions.
\newcommand{\StyleInstrShape}{\relax}

% ---- 5. running head -----------------------------------------------------------------
%   Measured: this book's running head is sans text in the body ink colour, and there is
%   NO hairline under it. Enable \StyleHeadRuletrue only if your book rules its head, and
%   then put the rule's measured colour in StyleHeadRule above.
\newif\ifStyleHeadRule
\StyleHeadRulefalse

% -- section heading. With the banner on this reproduces the book's exercise opener:
%      ----------------------------  (full-width grey hairline)
%      [ EXERCISES ]  3.6   The Chain Rule
%    Give an empty second argument if the book does not print the section title here.
\newcommand{\exbanner}[2]{%
  \par\needspace{6\baselineskip}\medskip
  \noindent
  \ifStyleBanner
    % hairline first, so the opaque box painted after it covers the overlap
    \rlap{\raisebox{\StyleBannerRaise}{\rlap{\color{StyleBannerRule}%
      \rule{\linewidth}{0.5pt}}}}%
    \colorbox{StyleBanner}{\textcolor{StyleBannerText}{\StyleSansFont\bfseries
      \fontsize{14}{16}\selectfont\,\StyleBannerLabel\,}}%
    \hspace{0.55em}%
    {\StyleSansFont\bfseries\fontsize{17}{18}\selectfont\color{StyleNumber}#1}%
    \ifx\relax#2\relax\else\hspace{0.7em}{\StyleSansFont\bfseries\large\color{StyleAccent}#2}\fi
  \else
    {\StyleSansFont\bfseries\large\color{StyleAccent}#1}%
    \ifx\relax#2\relax\else\quad{\StyleSansFont\large\color{StyleAccent}#2}\fi
    \par\smallskip
    \noindent\textcolor{StyleBannerRule}{\rule{\linewidth}{0.5pt}}%
  \fi
  \par\medskip
}

% -- the small blue sans run-in heading that titles a block of problems,
%    e.g. "Derivative Calculations", "Motion Along a Coordinate Line", "Economics".
\newcommand{\grouphead}[1]{%
  \par\needspace{5\baselineskip}\smallskip
  \noindent{\StyleSansFont\bfseries\small\color{StyleAccent}#1}\par\nobreak\smallskip
}

% -- the exercise-set instruction paragraph a group of problems depends on.
%    ALWAYS include it -- without it the reader cannot tell what is being asked.
\newcommand{\instr}[1]{%
  \par\needspace{3\baselineskip}\smallskip\noindent
  {\StyleInstrShape #1}\par\smallskip
}

% -- a numbered problem. Bold serif number, hanging indent, never stranded at a page foot.
\newcommand{\prob}[1]{%
  \par\needspace{4\baselineskip}\medskip\noindent
  \hangindent=1.75em\hangafter=1
  {\StyleBodyFont\bfseries#1}\hspace{0.45em}\ignorespaces
}

% This group has NO instruction paragraph in the book (page 151 goes straight from the

% the instruction a problem depends on", not "always print an instruction": inventing text
% the book does not have is the same class of error as dropping text it does.
\prob{23.} \textbf{Marginal cost}\quad Suppose that the dollar cost of producing $x$ washing
machines is $c(x)=2000+100x-0.1x^{2}$.
\begin{enumerate}
  \item Find the average cost per machine of producing the first 100 washing machines.
  \item Find the marginal cost when 100 washing machines are produced.
\end{enumerate}

\prob{75.} Suppose that functions $f$ and $g$ and their derivatives with respect to $x$
have the following values at $x=2$ and $x=3$.
\begin{dtable}
x & f(x) & g(x) & f'(x) & g'(x)\\[2pt]
\hline
2 & 8 & 2 & 1/3 & -3\\[2pt]
3 & 3 & -4 & 2\pi & 5
\end{dtable}
\noindent Find the derivatives with respect to $x$ of the following combinations at the
given value of $x$.
\begin{enumerate}
  \item $2f(x)$, \quad $x=2$
  \item $f(x)+g(x)$, \quad $x=3$
\end{enumerate}
